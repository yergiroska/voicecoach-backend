# VoiceCoach AI - backend
# Dependencias directas (las transitivas las resuelve pip).
# Las librerías de DB (PostgreSQL/pgvector) se añadirán en fases posteriores.

fastapi==0.139.2
uvicorn[standard]==0.51.0
pydantic-settings==2.14.2
python-dotenv==1.2.2

# Auth: verificación de ID tokens de Firebase contra las claves públicas de
# Google. Se usa google-auth (solo necesita el Project ID, público) en lugar
# del SDK Admin de Firebase, que exigiría una clave privada de servicio.
# El extra [requests] aporta el transporte HTTP google.auth.transport.requests.
google-auth[requests]==2.56.3

# Parseo de multipart/form-data: lo exige FastAPI para recibir archivos
# subidos (UploadFile) en POST /recordings.
python-multipart==0.0.32

# IA: transcripción de audio con Groq Whisper (modelo whisper-large-v3).
# Se usa el cliente asíncrono AsyncGroq porque el endpoint de subida es async.
# Arrastra httpx como transitiva, que además es lo que necesita
# fastapi.testclient.TestClient cuando se añadan tests.
groq==1.7.0

# ---- Persistencia: PostgreSQL (+ pgvector en la Fase 7) ----
#
# El extra [asyncio] arrastra greenlet, que es lo que permite a SQLAlchemy
# ejecutar su capa síncrona dentro de un event loop. Sin él, `create_async_engine`
# falla en el import.
sqlalchemy[asyncio]==2.0.52

# Driver asíncrono de PostgreSQL. Se elige asyncpg (y no psycopg) para que el
# acceso a la DB no bloquee el event loop, en coherencia con AsyncGroq y con el
# endpoint async de POST /recordings.
asyncpg==0.31.0

# Migraciones de schema. Se usa desde el minuto uno (no `create_all`) porque el
# schema tiene que poder replicarse en Supabase en la Fase 9 sin divergencias.
alembic==1.19.2
